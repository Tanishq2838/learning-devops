# REGEX — THE CORE OF TODAY'S LESSON
import re
LOG_PATTERN = re.compile(r'(\d+\.\d+\.\d+\.\d+) - - \[(.*?)\] "(\w+) (.*?) HTTP/1.1" (\d+) (\d+)')

# r'...'        raw string — backslashes passed through untouched, required for regex
# \d            any digit                  \d+   one or more digits
# \.            literal dot (escaped, since . alone means "any character")
# ( )           capture group — extracts the matched text afterward via .group(n)
# .*?           non-greedy "any characters" — stops at the FIRST match, not the last
# \[ \]         literal square brackets (escaped, since [ ] are special in regex)
# \w+           one or more "word characters" (letters/digits/underscore)
